cmake_minimum_required(VERSION 3.20)
include(${CMAKE_CURRENT_LIST_DIR}/../../../scripts/bench_clock_profile.cmake)
option(BENCH_DIAGNOSTICS "Functional UART feedback; never use this profile for energy capture" OFF)
set(IDF_TARGET esp32 CACHE STRING "Fixed measurement target")
if(BENCH_COMMON_CLOCK_160)
  set(PROJECT_VER "1.2.0")
  set(BENCH_ESP32_CPU_MHZ 160)
  set(BENCH_ESP32_DEFAULTS "${CMAKE_CURRENT_LIST_DIR}/sdkconfig.defaults.common160")
else()
  set(PROJECT_VER "1.1.0")
  set(BENCH_ESP32_CPU_MHZ 240)
  set(BENCH_ESP32_DEFAULTS "${CMAKE_CURRENT_LIST_DIR}/sdkconfig.defaults")
endif()

# IDF otherwise shares PROJECT_DIR/sdkconfig across independent build trees.
# A defaults file does not override values already present in that sdkconfig.
set(BENCH_ESP32_SDKCONFIG "${CMAKE_BINARY_DIR}/sdkconfig")
if(DEFINED SDKCONFIG AND NOT SDKCONFIG STREQUAL "")
  get_filename_component(BENCH_REQUESTED_SDKCONFIG "${SDKCONFIG}" ABSOLUTE)
  if(NOT BENCH_REQUESTED_SDKCONFIG STREQUAL BENCH_ESP32_SDKCONFIG)
    message(FATAL_ERROR "SDKCONFIG must be ${BENCH_ESP32_SDKCONFIG}; each clock profile requires its own build configuration")
  endif()
endif()
if(DEFINED SDKCONFIG_DEFAULTS AND NOT SDKCONFIG_DEFAULTS STREQUAL "")
  get_filename_component(BENCH_REQUESTED_DEFAULTS "${SDKCONFIG_DEFAULTS}" ABSOLUTE)
  if(NOT BENCH_REQUESTED_DEFAULTS STREQUAL BENCH_ESP32_DEFAULTS)
    message(FATAL_ERROR "SDKCONFIG_DEFAULTS must match the selected clock profile: ${BENCH_ESP32_DEFAULTS}")
  endif()
endif()
set(SDKCONFIG "${BENCH_ESP32_SDKCONFIG}" CACHE FILEPATH "Configuration isolated to this build tree" FORCE)
set(SDKCONFIG_DEFAULTS "${BENCH_ESP32_DEFAULTS}" CACHE STRING "Defaults for the selected benchmark profile" FORCE)
include($ENV{IDF_PATH}/tools/cmake/project.cmake)
project(energy_bench_esp32)
if(NOT CONFIG_ESP_DEFAULT_CPU_FREQ_MHZ EQUAL BENCH_ESP32_CPU_MHZ)
  message(FATAL_ERROR "sdkconfig CPU frequency does not match BENCH_CLOCK_PROFILE=${BENCH_CLOCK_PROFILE}; use an untouched build directory")
endif()
set(BENCH_IDF_VERSION "${IDF_VERSION_MAJOR}.${IDF_VERSION_MINOR}.${IDF_VERSION_PATCH}")
if(NOT BENCH_IDF_VERSION STREQUAL "5.5.4")
  message(FATAL_ERROR "This measurement profile requires ESP-IDF 5.5.4; found ${BENCH_IDF_VERSION}")
endif()
