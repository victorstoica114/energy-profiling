cmake_minimum_required(VERSION 3.20)
include(${CMAKE_CURRENT_LIST_DIR}/../../../scripts/bench_clock_profile.cmake)
option(BENCH_DIAGNOSTICS "Functional serial feedback; never use this profile for energy capture" OFF)
set(BENCH_DIAGNOSTIC_TRANSPORT "USB" CACHE STRING "Diagnostic transport: USB or UART0")
set_property(CACHE BENCH_DIAGNOSTIC_TRANSPORT PROPERTY STRINGS USB UART0)
if(NOT BENCH_DIAGNOSTIC_TRANSPORT MATCHES "^(USB|UART0)$")
  message(FATAL_ERROR "BENCH_DIAGNOSTIC_TRANSPORT must be USB or UART0")
endif()
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
if(NOT PICO_SDK_PATH AND DEFINED ENV{PICO_SDK_PATH})
  set(PICO_SDK_PATH "$ENV{PICO_SDK_PATH}")
endif()
if(NOT PICO_SDK_PATH)
  message(FATAL_ERROR "Set PICO_SDK_PATH to the pinned Pico SDK 2.2.0")
endif()
file(TO_CMAKE_PATH "${PICO_SDK_PATH}" PICO_SDK_PATH)
set(PICO_BOARD pico CACHE STRING "Board identity")
if(NOT PICO_BOARD STREQUAL "pico")
  message(FATAL_ERROR "This profile requires the non-wireless Raspberry Pi Pico board")
endif()
set(PICO_NO_PICOTOOL 1 CACHE BOOL "Use installed elf2uf2 separately if desired")
include(${PICO_SDK_PATH}/external/pico_sdk_import.cmake)
project(energy_bench_rp2040 C CXX ASM)
set(CMAKE_C_STANDARD 11)
# Applies to boot2 as well as application/SDK translation units: QSPI stays
# at 50 MHz (max_clock) or 40 MHz (common160) on the tested Marble Pico board.
add_compile_definitions(PICO_FLASH_SPI_CLKDIV=4)
pico_sdk_init()
if(BENCH_DIAGNOSTICS AND BENCH_DIAGNOSTIC_TRANSPORT STREQUAL "USB" AND NOT TARGET tinyusb_device)
  message(FATAL_ERROR "USB diagnostics require TinyUSB; set PICO_TINYUSB_PATH to the pinned diagnostic dependency")
endif()
if(NOT PICO_PLATFORM STREQUAL "rp2040")
  message(FATAL_ERROR "This benchmark target requires RP2040")
endif()
if(NOT PICO_SDK_VERSION_STRING STREQUAL "2.2.0")
  message(FATAL_ERROR "Expected Pico SDK 2.2.0, found ${PICO_SDK_VERSION_STRING}")
endif()
include(${CMAKE_CURRENT_LIST_DIR}/../../common/bench_sources.cmake)
add_executable(energy_bench_rp2040 platform_rp2040.c ${BENCH_COMMON_SOURCES})
target_include_directories(energy_bench_rp2040 PRIVATE ${BENCH_COMMON_INCLUDE_DIRS})
target_compile_definitions(energy_bench_rp2040 PRIVATE BENCH_BOARD_RP2040=1 PICO_STACK_SIZE=4096 BENCH_DIAGNOSTICS=$<BOOL:${BENCH_DIAGNOSTICS}> BENCH_COMMON_CLOCK_160=${BENCH_COMMON_CLOCK_160})
target_compile_options(energy_bench_rp2040 PRIVATE -O2 -fno-fast-math -ffp-contract=off -fno-lto -fstack-usage -Wall -Wextra)
target_link_libraries(energy_bench_rp2040 pico_stdlib hardware_clocks hardware_timer hardware_vreg m)
if(BENCH_DIAGNOSTICS)
  target_link_libraries(energy_bench_rp2040 hardware_flash hardware_sync)
endif()
if(BENCH_DIAGNOSTICS AND BENCH_DIAGNOSTIC_TRANSPORT STREQUAL "USB")
  target_compile_definitions(energy_bench_rp2040 PRIVATE BENCH_DIAGNOSTIC_USB=1)
  pico_enable_stdio_usb(energy_bench_rp2040 1)
else()
  target_compile_definitions(energy_bench_rp2040 PRIVATE BENCH_DIAGNOSTIC_USB=0)
  pico_enable_stdio_usb(energy_bench_rp2040 0)
endif()
if(BENCH_DIAGNOSTICS AND BENCH_DIAGNOSTIC_TRANSPORT STREQUAL "UART0")
  target_link_libraries(energy_bench_rp2040 hardware_uart)
endif()
pico_enable_stdio_uart(energy_bench_rp2040 0)
target_link_options(energy_bench_rp2040 PRIVATE "-Wl,-Map=${CMAKE_CURRENT_BINARY_DIR}/energy_bench_rp2040.map")
pico_add_extra_outputs(energy_bench_rp2040)
